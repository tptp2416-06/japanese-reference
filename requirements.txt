# Japanese Reference — MkDocs dependencies
#
# EVERY plugin named in mkdocs.yml must appear below. A CI build starts from an
# empty environment and installs only what this file lists — five plugins were
# missing here, so a Cloudflare Pages build would have failed at "plugin not
# installed" while `mkdocs serve` kept working locally off packages installed
# by hand months ago.
#
# Pinned exactly, not by range. Material for MkDocs reached maintenance mode in
# November 2025 and reaches end of life on 2026-11-05; after that there are no
# fixes. The risk is not that date — the build keeps working — it is an
# unpinned rebuild months from now pulling something incompatible into a site
# nobody has touched since. Material 9.7.5+ also caps MkDocs below 2.0 itself,
# because MkDocs 2.0 (Feb 2026) breaks plugins.
#
# Do NOT run `pip install --upgrade` without reading release notes first.
#
# To install: pip install -r requirements.txt
# To serve locally: mkdocs serve
# To build: mkdocs build

mkdocs==1.6.1
mkdocs-material==9.7.6

# Plugins referenced by mkdocs.yml. `search` ships with MkDocs and `tags` ships
# with Material, so neither needs a line here.
mkdocs-awesome-pages-plugin==2.10.1
mkdocs-git-revision-date-localized-plugin==1.5.1
mkdocs-glightbox==0.5.2
mkdocs-minify-plugin==0.8.0
mkdocs-section-index==0.3.12

# Markdown extensions referenced by mkdocs.yml (pymdownx.*)
pymdown-extensions==10.21.2

# Transitive, and pinned for the same reason as everything above.
#
# mkdocs-section-index has migrated its dependency to ProperDocs — the MkDocs
# 1.x community fork, started by a former MkDocs maintainer after the original
# owner stopped maintaining 1.x. So this arrives whether or not it is asked for,
# and unpinned it would drift silently.
#
# It is NOT what builds the site: `mkdocs build` still runs MkDocs 1.6.1. This
# line only stops a transitive dependency moving underneath a pinned build.
#
# Note for whoever reads `pip show properdocs`: the Author-email field says Tom
# Christie because it was inherited from the package it forked. He is not
# involved in ProperDocs. It is not an endorsement.
properdocs==1.6.7
