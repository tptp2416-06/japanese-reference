# Japanese Reference — MkDocs dependencies
#
# EVERY plugin named in mkdocs.yml must appear below. A CI build starts from an
# empty environment and installs only what this file lists — five plugins were
# missing here, so a Cloudflare Pages build would have failed at "plugin not
# installed" while `mkdocs serve` kept working locally off packages installed
# by hand months ago.
#
# Pinned exactly, not by range. Material for MkDocs reached maintenance mode in
# November 2025 and reaches end of life on 2026-11-05; after that there are no
# fixes. The risk is not that date — the build keeps working — it is an
# unpinned rebuild months from now pulling something incompatible into a site
# nobody has touched since. Material 9.7.5+ also caps MkDocs below 2.0 itself,
# because MkDocs 2.0 (Feb 2026) breaks plugins.
#
# Do NOT run `pip install --upgrade` without reading release notes first.
#
# To install: pip install -r requirements.txt
# To serve locally: mkdocs serve
# To build: mkdocs build

mkdocs==1.6.1
mkdocs-material==9.7.6
